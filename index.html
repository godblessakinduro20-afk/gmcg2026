<script>

let contestants = JSON.parse(localStorage.getItem('gmcg_contestants')) || [];

let votes = JSON.parse(localStorage.getItem('gmcg_votes')) || [];

let currentContestant = null;

let loggedInContestant = null;

(function(){ emailjs.init("YOUR_PUBLIC_KEY"); })();

function sendEmail(to, subject, body){

  emailjs.send("YOUR_SERVICE_ID", "YOUR_TEMPLATE_ID", {

    to_email: to, from_name: "GMCG 2026", subject: subject, message: body, reply_to: "YOUR_ADMIN_EMAIL"

  });

}

function saveData(){

  localStorage.setItem('gmcg_contestants', JSON.stringify(contestants));

  localStorage.setItem('gmcg_votes', JSON.stringify(votes));

}

function acceptPolicy(){ if(document.getElementById('acceptPolicy').checked){ document.getElementById('policyPopup').style.display='none'; }else{ alert('You must accept the Privacy Policy to continue'); }}

function showPage(pageId){ document.querySelectorAll('.page').forEach(p => p.classList.remove('active')); document.getElementById(pageId).classList.add('active'); window.scrollTo(0,0); }

function switchTab(tab){

  if(tab=='signup'){ document.getElementById('signup-form').style.display='block'; document.getElementById('login-form').style.display='none'; document.getElementById('tab-signup').classList.add('active-tab'); document.getElementById('tab-login').classList.remove('active-tab'); }

  else{ document.getElementById('signup-form').style.display='none'; document.getElementById('login-form').style.display='block'; document.getElementById('tab-login').classList.add('active-tab'); document.getElementById('tab-signup').classList.remove('active-tab'); }

}

function contestantSignup(){

  let name = document.getElementById('cname').value;

  let email = document.getElementById('cemail').value;

  let pass = document.getElementById('cpass').value;

  if(contestants.find(c => c.email == email)){ alert('This email already registered. Please Login'); return; }

  let ref = 'GMCG' + Math.floor(Math.random()*90000 + 10000);

  let newContestant = {name:name, email:email, pass:pass, phone:document.getElementById('cphone').value, category:document.getElementById('ccategory').value, bio:"Bio data will be updated by admin soon.", ref:ref, votes:0, earnings:0, photo:'', trend:'up'};

  contestants.push(newContestant);

  saveData();

  loginContestant(newContestant);

  alert('Account Created Successfully!');

}

function contestantLogin(){

  let email = document.getElementById('clogin-email').value;

  let pass = document.getElementById('clogin-pass').value;

  let c = contestants.find(x => x.email == email && x.pass == pass);

  if(c){ loginContestant(c); }

  else{ alert('Invalid Email or Password'); }

}

function loginContestant(c){

  loggedInContestant = c;

  currentContestant = contestants.indexOf(c);

  document.getElementById('dash-name').innerText = c.name;

  document.getElementById('dash-email').innerText = c.email;

  document.getElementById('dash-cat').innerText = c.category;

  document.getElementById('dash-bio').innerText = c.bio;

  document.getElementById('ref-code').innerText = c.ref;

  document.getElementById('contestant-photo').innerHTML = c.photo? `<img src="${c.photo}">` : '?';

  if(!document.querySelector(`#vote-contestant option[value="${c.ref}"]`)){

    let opt = document.createElement('option');

    opt.text = c.name + ' - ' + c.ref;

    opt.value = c.ref;

    document.getElementById('vote-contestant').add(opt);

  }

  updateMyVotesStatus();

  startTrendAnimation();

  showPage('contestant-dash');

}

function logoutContestant(){

  loggedInContestant = null;

  currentContestant = null;

  showPage('home');

}

function voterSubmit(){ document.getElementById('voter-name').innerText = document.getElementById('vname').value; showPage('vote'); }

function showContestantInfo(){ let ref = document.getElementById('vote-contestant').value; let c = contestants.find(x => x.ref == ref); if(c){ document.getElementById('selected-contestant').style.display='block'; document.getElementById('vote-cname').innerText = c.name; document.getElementById('vote-ccat').innerText = c.category; document.getElementById('vote-cbio').innerText = c.bio; document.getElementById('vote-cref').innerText = c.ref; document.getElementById('vote-photo').innerHTML = c.photo? `<img src="${c.photo}">` : '?'; }}

function copyRef(){ navigator.clipboard.writeText(document.getElementById('ref-code').innerText); alert('Reference code copied!'); }

function copyVoteRef(){ navigator.clipboard.writeText(document.getElementById('vote-cref').innerText); alert('Reference code copied!'); }

function checkProof(){ let file = document.getElementById('vote-proof').files[0]; document.getElementById('submit-vote-btn').disabled =!file; }

function submitVote(){ let voter = document.getElementById('voter-name').innerText; let ref = document.getElementById('vote-contestant').value; let amount = parseInt(document.getElementById('vote-amount').value); let payref = document.getElementById('vote-payref').value; let voteCount = Math.floor(amount / 100); votes.push({voter:voter, ref:ref, amount:amount, votes:voteCount, payref:payref, status:'Pending'}); let c = contestants.find(x => x.ref == ref); if(c){ c.votes += voteCount; c.earnings += amount; } saveData(); sendEmail("YOUR_ADMIN_EMAIL", 'New Vote Submitted', `${voter} voted for ${ref} - Amount: ₦${amount} - Votes: ${voteCount}`); updateMyVotesStatus(); document.getElementById('live-votes').innerText = votes.reduce((a,b)=>a+b.votes,0); updateAdmin(); alert(`Vote Submitted For Approval! ${voteCount} votes. Waiting for Admin approval.`); }

function updateMyVotesStatus(){ if(!loggedInContestant) return; let myVotes = votes.filter(v => v.voter == loggedInContestant.name); let html = '<h3>My Vote Status</h3>'; if(myVotes.length == 0){ html += '<p>No votes yet</p>'; } else{ myVotes.forEach(v => { let badge = v.status == 'Approved'? 'status-approved' : 'status-pending'; html += `<p>${v.ref}: ₦${v.amount} - <span class="status-badge ${badge}">${v.status}</span></p>`; }); } document.getElementById('my-votes-status').innerHTML = html; }

function startTrendAnimation(){ setInterval(() => { if(currentContestant!== null){ let trend = Math.random() > 0.5? 'up' : 'down'; contestants[currentContestant].trend = trend; if(document.getElementById('trend-arrow')){ document.getElementById('trend-arrow').className = 'trend ' + trend; document.getElementById('trend-arrow').innerText = trend == 'up'? '↑' : '↓'; } }, 3000); }

function updateAdmin(){ document.getElementById('total-contestants').innerText = contestants.length; document.getElementById('total-votes').innerText = votes.reduce((a,b)=>a+b.votes,0); document.getElementById('total-earnings').innerText = votes.reduce((a,b)=>a+b.amount,0); let cTable = document.getElementById('contestants-table'); cTable.innerHTML = '<tr><th>Photo</th><th>Name</th><th>Email</th><th>Ref Code</th><th>Votes</th><th>Earnings</th><th>Edit</th></tr>'; contestants.forEach((c,i) => { cTable.innerHTML += `<tr><td><div class="profile-pic" style="width:50px;height:50px;">${c.photo? `<img src="${c.photo}">` : '?'}</div><input type="file" onchange="uploadPhoto(${i}, this)"></td><td>${c.name}</td><td>${c.email}</td><td>${c.ref}</td><td>${c.votes}</td><td>₦${c.earnings}</td><td><button class="btn small" onclick="editBio(${i})">Edit Bio</button></td></tr>`; }); let vTable = document.getElementById('votes-table'); vTable.innerHTML = '<tr><th>Voter</th><th>Contestant</th><th>Amount</th><th>Votes</th><th>Status</th><th>Action</th></tr>'; votes.forEach((v,i) => { let badge = v.status == 'Approved'? 'status-approved' : 'status-pending'; vTable.innerHTML += `<tr><td>${v.voter}</td><td>${v.ref}</td><td>₦${v.amount}</td><td>${v.votes}</td><td><span class="status-badge ${badge}">${v.status}</span></td><td>${v.status=='Pending'? `<button class="btn small" onclick="approveVote(${i})">Approve</button>` : 'Approved'}</td></tr>`; }); }

function uploadPhoto(index, input){ let file = input.files[0]; let reader = new FileReader(); reader.onload = function(e){ contestants[index].photo = e.target.result; saveData(); updateAdmin(); if(currentContestant == index){ document.getElementById('contestant-photo').innerHTML = `<img src="${e.target.result}">`; } reader.readAsDataURL(file); }

function editBio(index){ let newBio = prompt("Enter correct bio data for " + contestants[index].name, contestants[index].bio); if(newBio){ contestants[index].bio = newBio; saveData(); if(currentContestant == index){ document.getElementById('dash-bio').innerText = newBio; } updateAdmin(); }}

function approveVote(index){ votes[index].status = 'Approved'; saveData(); sendEmail("YOUR_ADMIN_EMAIL", 'Vote Approved', `Vote of ₦${votes[index].amount} for ${votes[index].ref} has been approved`); alert('Vote Approved!'); updateAdmin(); updateMyVotesStatus(); }

function login(){ let email = document.getElementById('email').value; let pass = document.getElementById('pass').value; if(email && pass){ showPage('admin-panel'); updateAdmin(); } else{ alert('Enter Email and Password'); } }

</script>
